cmake_minimum_required(VERSION 3.16)
project(grok_gadgets_esp32_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE lib/GrokGadgets/src)
target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME core COMMAND core_tests)

set(ARDUINOJSON_INCLUDE "${CMAKE_SOURCE_DIR}/.pio/libdeps/atoms3-lite-usb/ArduinoJson/src" CACHE PATH "ArduinoJson 6.21.5 include directory")
if(NOT EXISTS "${ARDUINOJSON_INCLUDE}/ArduinoJson.h")
 message(FATAL_ERROR "Run .venv/bin/pio pkg install before host tests")
endif()
add_executable(device_tests tests/device_tests.cpp)
target_include_directories(device_tests PRIVATE lib/GrokGadgets/src "${ARDUINOJSON_INCLUDE}")
target_compile_options(device_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME device COMMAND device_tests)

# Targets below compile Arduino sketches against the simulated board in tests/stubs.
function(host_sketch name source)
 add_executable(${name} ${source})
 target_include_directories(${name} PRIVATE tests/stubs lib/GrokGadgets/src "${ARDUINOJSON_INCLUDE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
endfunction()
host_sketch(session_tests tests/session_tests.cpp)
add_test(NAME session COMMAND session_tests)
host_sketch(firmware_tests tests/firmware_tests.cpp)
add_test(NAME firmware_simulation COMMAND firmware_tests)
host_sketch(host_firmware tests/host_firmware.cpp)
host_sketch(transcript_c124 tests/transcript.cpp)
target_compile_definitions(transcript_c124 PRIVATE GROK_EXAMPLE="${CMAKE_SOURCE_DIR}/src/main.cpp")
host_sketch(transcript_led_button tests/transcript.cpp)
target_compile_definitions(transcript_led_button PRIVATE GROK_EXAMPLE="${CMAKE_SOURCE_DIR}/examples/led-button/main.cpp")
